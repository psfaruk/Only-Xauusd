   159→const dhakaDateTime = new Intl.DateTimeFormat('bn-BD', {
   160→  timeZone: 'Asia/Dhaka',
   161→  dateStyle: 'medium',
   162→  timeStyle: 'short',
   163→});
   164→
   165→function fmtDhaka(iso: string): string {
   166→  try {
   167→    return dhakaDateTime.format(new Date(iso));
   168→  } catch {
   169→    return iso;
   170→  }
   171→}
   172→
   173→function fmtUptime(seconds: number): string {
   174→  if (seconds < 60) return `${seconds}s`;
   175→  const m = Math.floor(seconds / 60);
   176→  const s = seconds % 60;
   177→  if (m < 60) return `${m}m ${s}s`;
   178→  const h = Math.floor(m / 60);
   179→  return `${h}h ${m % 60}m`;
   180→}

   183→  runtime: 'রানটাইম',

   185→  glm: 'GLM',
   186→  mt5: 'MT5',
   187→  database: 'ডেটাবেস',

   189→
   190→function useDhakaClock(): string | null {
   191→  const [now, setNow] = useState<string | null>(null);
   192→  useEffect(() => {
   193→    const fmt = new Intl.DateTimeFormat('bn-BD', {
   194→      timeZone: 'Asia/Dhaka',
   195→      hour: '2-digit',
   196→      minute: '2-digit',
   197→      second: '2-digit',
   198→      hour12: true,
   199→    });
   200→    const tick = () => setNow(fmt.format(new Date()));
   201→    tick();
   202→    const id = setInterval(tick, 1000);
   203→    return () => clearInterval(id);
   204→  }, []);
   205→  return now;
   206→}

   208→function StatusIcon({ status }: { status: 'pass' | 'warn' | 'fail' | 'active' | 'planned' | string }) {
   209→  if (status === 'pass' || status === 'active' || status === 'verified')
   210→    return <CheckCircle2 className="h-4 w-4 text-emerald-400" aria-hidden />;
   211→  if (status === 'fail') return <XCircle className="h-4 w-4 text-red-400" aria-hidden />;
   212→  return <AlertTriangle className="h-4 w-4 text-amber-400" aria-hidden />;
   213→}
   214→
   215→function levelBadgeClass(level: string): string {
   216→  switch (level) {
   217→    case 'error':
   218→      return 'border-red-500/40 bg-red-500/10 text-red-400';
   219→    case 'warn':
   220→      return 'border-amber-500/40 bg-amber-500/10 text-amber-400';
   221→    case 'info':
   222→      return 'border-zinc-500/40 bg-zinc-500/10 text-zinc-300';
   223→    default:
   224→      return 'border-zinc-700 bg-zinc-800/50 text-zinc-500';
   225→  }
   226→}
